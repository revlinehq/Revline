function PixelIris({children,id,seconds=.83,power=.72,edgeLead=.035,edgeTrail=.085}) {
 const {width,height}=useScene();
 const rings=Array.from({length:34},()=>[]);
 for(let y=0;y<36;y++)for(let x=0;x<64;x++){
  const dx=(x-31.5)*30,dy=(y-17.5)*30;
  const noise=((x*37+y*61)%13-6)*9;
  const index=Math.min(33,Math.max(0,Math.floor((Math.hypot(dx,dy)+noise)/35)));
  rings[index].push('M'+(x*30)+' '+(y*30)+'h30v30h-30Z');
 }
 const ref=useTimeline((tl,el)=>{
  tl.to('.mask-full',{opacity:1,duration:.001},seconds+.03);
  rings.forEach((_,i)=>{
   const at=.001+Math.pow(i/33,power)*seconds;
   tl.to('.mask-'+i,{opacity:1,duration:.001},at);
   tl.to('.edge-'+i,{opacity:1,duration:.001},Math.max(.001,at-edgeLead))
    .to('.edge-'+i,{opacity:0,duration:.001},at+edgeTrail);
  });
 });
 return <div ref={ref} style={FILL}>
  <svg style={{...FILL,width:'100%',height:'100%'}} viewBox="0 0 1920 1080">
   <defs><mask id={id} maskUnits="userSpaceOnUse" x="0" y="0" width="1920" height="1080"><rect width="1920" height="1080" fill="#000"/>{rings.map((paths,i)=><path key={i} className={'mask-'+i} d={paths.join('')} fill="#fff" style={{opacity:0}}/>)}<rect className="mask-full" width="1920" height="1080" fill="#fff" style={{opacity:0}}/></mask></defs>
   <foreignObject x="0" y="0" width="1920" height="1080" mask={'url(#'+id+')'}><div style={{position:'relative',width:1920,height:1080,overflow:'hidden'}}>{children}</div></foreignObject>
   {rings.map((paths,i)=><path key={i} className={'edge-'+i} d={paths.join('')} fill={LIME} style={{opacity:0}}/>)}
  </svg>
 </div>;
}
function PixelGlobe() {
 const rows=30,cols=56;
 const cells=[];
 for(let r=0;r<rows;r++)for(let c=0;c<cols;c++)cells.push({r,c});
 const ref=useTimeline((tl,el)=>{
  const samples=el.querySelectorAll('.texture-shape');
  const nodes=el.querySelectorAll('.globe-cell');
  const point=el.createSVGPoint();
  const black=cells.map(({r,c})=>{
   point.x=(((c/cols*2+.5)%1)*1.5-.25)*340;point.y=r/(rows-1)*390;
   return Array.from(samples).some(path=>path.isPointInFill(point));
  });
  const state={angle:.28};
  const project=(lon,lat)=>{
   const radius=407*Math.cos(lat*.35);
   const z=Math.cos(lon);
   return [420+Math.sin(lon)*radius,450+Math.sin(lat)*455*(.78+.22*z)];
  };
  const draw=()=>{
   cells.forEach(({r,c},i)=>{
    const lon=c/cols*Math.PI*2+state.angle,lat=-1.37+r/rows*2.74;
    const node=nodes[i];if(!node)return;
    if(Math.cos(lon)<.02){node.style.display='none';return;}
    node.style.display='';
    const dlon=Math.PI*2/cols*.84,dlat=2.74/rows*.84;
    const points=[project(lon-dlon/2,lat-dlat/2),project(lon+dlon/2,lat-dlat/2),project(lon+dlon/2,lat+dlat/2),project(lon-dlon/2,lat+dlat/2)];
    node.setAttribute('d','M'+points.map(p=>p.map(v=>v.toFixed(1)).join(',')).join('L')+'Z');
    node.setAttribute('fill',black[i]?'#050505':LIME);
   });
  };
  draw();
  tl.to(state,{angle:-1.45,duration:8.32,ease:'none',onUpdate:draw},.001);
 });
 return <svg ref={ref} viewBox="0 0 840 900" width="840" height="900" style={{overflow:'visible'}}>
  <defs>{BODY_PATHS.map((p,i)=><path className="texture-shape" key={i} d={p.d}/>)}</defs>
  {cells.map((p,i)=><path key={i} className="globe-cell" fill={LIME}/>)}
  <path d="M15 25V9H31 M809 9H825V25 M15 875V891H31 M809 891H825V875" fill="none" stroke="#242424" strokeWidth="3"/>
 </svg>;
}
